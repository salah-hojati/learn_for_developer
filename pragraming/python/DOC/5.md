### dataclass

به جای نوشتن کلاس دستی، از `@dataclass` استفاده کن:

</div>


```python
from dataclasses import dataclass, field

@dataclass
class Student:
    name: str
    age: int
    grades: list[float] = field(default_factory=list)

    @property
    def average(self) -> float:
        if not self.grades:
            return 0.0
        return sum(self.grades) / len(self.grades)


s = Student("علی", 15)
s.grades.append(18.5)
s.grades.append(19.0)

print(s.average)  # 18.75
```

### `app/database.py`

```python
import sqlite3
from pathlib import Path

DB_PATH = Path("data") / "app.db"

def get_connection() -> sqlite3.Connection:
    DB_PATH.parent.mkdir(exist_ok=True)
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row  # دسترسی با نام ستون
    return conn

def init_db() -> None:
    with get_connection() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS students (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                age INTEGER NOT NULL
            )
        """)
        conn.execute("""
            CREATE TABLE IF NOT EXISTS grades (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                student_id INTEGER NOT NULL,
                course TEXT NOT NULL,
                score REAL NOT NULL,
                FOREIGN KEY (student_id) REFERENCES students(id)
            )
        """)
```

### `app/services.py`

```python
from app.database import get_connection
from app.models import Student, Grade

def add_student(name: str, age: int) -> int:
    with get_connection() as conn:
        cursor = conn.execute(
            "INSERT INTO students (name, age) VALUES (?, ?)",
            (name, age),
        )
        return cursor.lastrowid

def get_student(student_id: int) -> Student | None:
    with get_connection() as conn:
        row = conn.execute(
            "SELECT * FROM students WHERE id = ?", (student_id,)
        ).fetchone()
    if row is None:
        return None
    return Student(id=row["id"], name=row["name"], age=row["age"])

def list_students() -> list[Student]:
    with get_connection() as conn:
        rows = conn.execute("SELECT * FROM students").fetchall()
    return [Student(id=r["id"], name=r["name"], age=r["age"]) for r in rows]

def add_grade(student_id: int, course: str, score: float) -> int:
    if not 0 <= score <= 20:
        raise ValueError("نمره باید بین 0 و 20 باشد")
    with get_connection() as conn:
        cursor = conn.execute(
            "INSERT INTO grades (student_id, course, score) VALUES (?, ?, ?)",
            (student_id, course, score),
        )
        return cursor.lastrowid

def student_average(student_id: int) -> float:
    with get_connection() as conn:
        row = conn.execute(
            "SELECT AVG(score) AS avg FROM grades WHERE student_id = ?",
            (student_id,),
        ).fetchone()
    return row["avg"] or 0.0
```

### `app/main.py`

```python
from app import services
from app.database import init_db

def menu() -> None:
    init_db()
    while True:
        print("\n1) افزودن دانشجو")
        print("2) افزودن نمره")
        print("3) نمایش دانشجوها")
        print("4) میانگین دانشجو")
        print("5) خروج")

        choice = input("انتخاب: ")

        try:
            if choice == "1":
                name = input("اسم: ")
                age = int(input("سن: "))
                sid = services.add_student(name, age)
                print(f"دانشجو با شناسه {sid} ثبت شد.")

            elif choice == "2":
                sid = int(input("شناسه دانشجو: "))
                course = input("درس: ")
                score = float(input("نمره: "))
                services.add_grade(sid, course, score)
                print("نمره ثبت شد.")

            elif choice == "3":
                for s in services.list_students():
                    print(f"[{s.id}] {s.name} - {s.age}")

            elif choice == "4":
                sid = int(input("شناسه دانشجو: "))
                print("میانگین:", services.student_average(sid))

            elif choice == "5":
                break
            else:
                print("گزینه نامعتبر.")
        except ValueError as e:
            print("خطا:", e)

if __name__ == "__main__":
    menu()
```

from fastapi import FastAPI, HTTPException
from app import services
from app.database import init_db

app = FastAPI()

@app.on_event("startup")
def startup():
    init_db()

@app.get("/students")
def get_students():
    return [s.__dict__ for s in services.list_students()]

@app.get("/students/{sid}")
def get_student(sid: int):
    s = services.get_student(sid)
    if s is None:
        raise HTTPException(status_code=404, detail="پیدا نشد")
    return s.__dict__

@app.post("/students")
def create_student(name: str, age: int):
    sid = services.add_student(name, age)
    return {"id": sid}

# استفاده از SSH SOCKS Proxy در ترمینال لینوکس

اسکریپت شما فقط یه **SOCKS5 proxy** روی `127.0.0.1:9090` بالا می‌آره. برای اینکه ترافیک ترمینال از این تونل رد بشه، چند روش داری که از ساده‌ترین تا کامل‌ترین رو برات می‌نویسم:

---

## ⚠️ اول یه هشدار امنیتی

پایین فایل یه پسورد به صورت کامنت گذاشتی:
```
#    erw6iFY52jnUmfsz
```
اگه این پسورد مربوط به سرور `82.115.8.190` (root) هست، **فوراً عوضش کن** و از این فایل پاکش کن. حتی به صورت کامنت هم نباید بمونه.

---

## روش ۱: متغیرهای محیطی (ساده‌ترین)

بعد از اجرای `./ssh-proxy.sh start`، تو همون ترمینال:

</div>


```bash
export ALL_PROXY="socks5://127.0.0.1:9090"
export http_proxy="socks5h://127.0.0.1:9090"
export https_proxy="socks5h://127.0.0.1:9090"
export HTTP_PROXY="$http_proxy"
export HTTPS_PROXY="$https_proxy"
```
<div dir="rtl" style="text-align: right; font-family: 'B Nazanin', Tahoma, sans-serif;">


**نکته مهم:** از `socks5h://` استفاده کن (حرف h اضافه) تا **DNS هم از طریق تونل** resolve بشه — وگرنه DNS leak داری و بعضی سایت‌ها کار نمی‌کنن.

این روش کار می‌کنه با: `curl`, `wget`, `git`, `pip`, `npm`, `apt`, و بیشتر ابزارهایی که متغیرهای env رو می‌شناسن.

برای تست:
</div>


```bash
curl -s https://ifconfig.me ; echo
# باید IP سرور 82.115.8.190 رو نشون بده
```
<div dir="rtl" style="text-align: right; font-family: 'B Nazanin', Tahoma, sans-serif;">


برای اینکه موقت نباشه، این خطوط رو به `~/.bashrc` یا `~/.zshrc` اضافه کن.

---

## روش ۲: proxychains-ng (کامل‌ترین راه — پیشنهاد من)

این ابزار هر برنامه‌ای رو مجبور می‌کنه از SOCKS بگذره، حتی اونایی که env vars رو نمی‌فهمن (مثل `ssh`, `nc`, `ping` با محدودیت, `nmap`, ...).

### نصب
</div>


```bash
# Debian/Ubuntu
sudo apt install proxychains4

# Arch
sudo pacman -S proxychains-ng

# Fedora
sudo dnf install proxychains-ng
```

### تنظیم

فایل `/etc/proxychains4.conf` (یا `~/.proxychains/proxychains.conf`) رو ویرایش کن:

```ini
strict_chain
proxy_dns
remote_dns_subnet 224
tcp_read_time_out 15000
tcp_connect_time_out 8000
quiet_mode

[ProxyList]
socks5 127.0.0.1 9090
```

### استفاده

```bash
proxychains4 curl https://ifconfig.me
proxychains4 git clone https://github.com/...
proxychains4 wget https://example.com/file
proxychains4 apt update        # فقط اگه DNS هم ریدایرکت بشه
```

**تیپ:** یه alias بذار تو `~/.bashrc`:
```bash
alias px='proxychains4'
```
بعد راحت: `px curl ifconfig.me`

## روش ۳: برای ابزارهای خاص

### git
</div>


```bash
git config --global http.proxy socks5h://127.0.0.1:9090
git config --global https.proxy socks5h://127.0.0.1:9090
# برگردوندن:
git config --global --unset http.proxy
git config --global --unset https.proxy
```

### apt (Debian/Ubuntu)
```bash
sudo nano /etc/apt/apt.conf.d/95proxies
```
بنویس:
```
Acquire::http::Proxy "socks5h://127.0.0.1:9090";
Acquire::https::Proxy "socks5h://127.0.0.1:9090";
```

### curl / wget فقط برای یه دستور
```bash
curl --socks5-hostname 127.0.0.1:9090 https://ifconfig.me
wget -e use_proxy=yes -e https_proxy=socks5h://127.0.0.1:9090 https://example.com
```

### ssh از داخل تونل (اگه سرور دیگه‌ای هم داری)
```bash
ssh -o ProxyCommand='nc -X 5 -x 127.0.0.1:9090 %h %p' user@another-host
```

## روش ۴: یک تابع کمکی به فایل اضافه کن

یه `case` جدید به اسکریپتت اضافه کن که env رو فقط تو ترمینال فعلی ست کنه:
</div>


```bash
# بعد از start)
start)
    start_tunnel
    ;;
```

یه گزینه `env` اضافه کن:

```bash
    env)
        if check_tunnel; then
            echo "export ALL_PROXY=socks5://127.0.0.1:$LOCAL_PORT"
            echo "export http_proxy=socks5h://127.0.0.1:$LOCAL_PORT"
            echo "export https_proxy=socks5h://127.0.0.1:$LOCAL_PORT"
            echo "export HTTP_PROXY=socks5h://127.0.0.1:$LOCAL_PORT"
            echo "export HTTPS_PROXY=socks5h://127.0.0.1:$LOCAL_PORT"
        else
            echo "Tunnel not running!" >&2
            exit 1
        fi
        ;;
```
<div dir="rtl" style="text-align: right; font-family: 'B Nazanin', Tahoma, sans-serif;">


بعد تو ترمینال:
</div>


```bash
eval "$(./ssh-proxy.sh env)"
```
<div dir="rtl" style="text-align: right; font-family: 'B Nazanin', Tahoma, sans-serif;">


اینطوری هر بار که تونل رو استارت زدی، فقط یه دستور کافیه.

---

## ⚠️ یه نکته درباره‌ی اسکریپت خودت

توی `ensure_routing` از `$WIRED_INTERFACE` استفاده کردی ولی **هیچ‌جا تعریفش نکردی**. این باعث می‌شه:
</div>


```bash
if echo "$CURRENT_ROUTE" | grep -q "$WIRED_INTERFACE"; then
```
<div dir="rtl" style="text-align: right; font-family: 'B Nazanin', Tahoma, sans-serif;">


عملاً با `grep -q ""` اجرا بشه که **همیشه موفق**ه — یعنی همیشه می‌خواد روت رو "fix" کنه. باید اولش اینو تعریف کنی:
</div>


```bash
WIRED_INTERFACE=$(ip route | grep "default" | grep -E "eth[0-9]|enp[0-9]s[0-9]" | awk '{print $5}' | head -1)
```
<div dir="rtl" style="text-align: right; font-family: 'B Nazanin', Tahoma, sans-serif;">


اگه کارت شبکه‌ی سیمی نداری، این متغیر خالی می‌مونه و باید یه چک `[ -n "$WIRED_INTERFACE" ]` هم اضافه کنی.

## خلاصه پیشنهادی

1. **پسورد توی فایل رو حذف کن و عوضش کن.**
2. برای کار روزمره: `proxychains4` نصب کن — ساده‌ترین و کامل‌ترین.
3. برای ابزارهایی که env می‌فهمن: `eval "$(./ssh-proxy.sh env)"`.
4. DNS رو حتماً از تونل بفرست (`socks5h://` یا `proxy_dns` در proxychains) وگرنه نشتی داری.
